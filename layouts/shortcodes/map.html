{{/* Embedded OpenStreetMap with a marker, plus links out for directions. OSM rather than Google Maps: no API key, no tracking cookies, and no account needed. The bbox is computed from the point so callers only pass coordinates. Usage: {{< map lat="41.2453" lon="-75.8822" label="Rodano's" >}} */}} {{- $lat := float (.Get "lat") -}} {{- $lon := float (.Get "lon") -}} {{- $pad := float (.Get "pad" | default 0.0035) -}} {{- $label := .Get "label" | default "Meeting location" -}} {{- $bbox := printf "%.5f,%.5f,%.5f,%.5f" (sub $lon $pad) (sub $lat (div $pad 2.0)) (add $lon $pad) (add $lat (div $pad 2.0)) -}}
